cmake_minimum_required(VERSION 3.16)
project(breeze-win VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)

set(QT_MAJOR_VERSION 6)
set(QT_NO_CREATE_VERSIONLESS_TARGETS ON)
set(QT_NO_CREATE_VERSIONLESS_FUNCTIONS ON)
find_package(Qt6 REQUIRED CONFIG COMPONENTS Core Gui DBus Widgets)
find_package(KF6 REQUIRED COMPONENTS CoreAddons ColorScheme Config GuiAddons I18n IconThemes)
find_package(KDecoration3 REQUIRED)

add_subdirectory(libbreezecommon)
add_subdirectory(kdecoration)
